export const AssetStudioModal: React.FC<AssetStudioModalProps> = ({ isOpen, onClose }) => {
  const [copiedId, setCopiedId] = useState<string | null>(null);
  const [selectedAssetId, setSelectedAssetId] = useState<string>(ASSET_DEFINITIONS[0].id);
  const [isProcessing, setIsProcessing] = useState<boolean>(false);
  const [toastMessage, setToastMessage] = useState<string | null>(null);
  const fileInputRef = useRef<HTMLInputElement>(null);

  const selectedAsset = ASSET_DEFINITIONS.find(a => a.id === selectedAssetId) || ASSET_DEFINITIONS[0];

  const showToast = (msg: string) => {
    setToastMessage(msg);
    setTimeout(() => setToastMessage(null), 3000);
  };

  const copyPrompt = (asset: AssetDef) => {
    playSound('click');
    navigator.clipboard.writeText(asset.prompt);
    setCopiedId(asset.id);
    showToast(`Đã copy prompt cho "${asset.nameVi}"! Hãy dán vào Gemini Pro.`);
    setTimeout(() => setCopiedId(null), 2000);
  };

  // HTML5 Canvas Background Remover: Turn pure/near white into transparent PNG
  const processAndUploadImage = async (fileOrBlob: Blob, filename: string) => {
    setIsProcessing(true);
    try {
      const img = new Image();
      const objectUrl = URL.createObjectURL(fileOrBlob);
      await new Promise((resolve, reject) => {
        img.onload = resolve;
        img.onerror = reject;
        img.src = objectUrl;
      });

      const canvas = document.createElement('canvas');
      canvas.width = img.width;
      canvas.height = img.height;
      const ctx = canvas.getContext('2d');
      if (!ctx) throw new Error('Không thể tạo canvas');

      ctx.drawImage(img, 0, 0);
      const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);
      const data = imgData.data;
      const w = canvas.width;
      const h = canvas.height;

      // BFS Flood Fill from edges for near-white pixels (threshold 238)
      const threshold = 238;
      const visited = new Uint8Array(w * h);
      const queue: number[] = [];

      const pushIfBg = (x: number, y: number) => {
        if (x < 0 || x >= w || y < 0 || y >= h) return;
        const idx = y * w + x;
        if (visited[idx]) return;
        const p = idx * 4;
        if (data[p] >= threshold && data[p+1] >= threshold && data[p+2] >= threshold) {
          visited[idx] = 1;
          queue.push(idx);
        }
      };

      for (let x = 0; x < w; x++) {
        pushIfBg(x, 0);
        pushIfBg(x, h - 1);
      }
      for (let y = 0; y < h; y++) {
        pushIfBg(0, y);
        pushIfBg(w - 1, y);
      }

      let head = 0;
      while (head < queue.length) {
        const curr = queue[head++];
        const x = curr % w;
        const y = Math.floor(curr / w);
        const p = curr * 4;
        data[p + 3] = 0; // Transparent!

        pushIfBg(x + 1, y);
        pushIfBg(x - 1, y);
        pushIfBg(x, y + 1);
        pushIfBg(x, y - 1);
      }

      ctx.putImageData(imgData, 0, 0);
      const base64Png = canvas.toDataURL('image/png');

      // Upload to dev server
      const res = await fetch('/api/save-asset', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ filename, base64: base64Png })
      });

      if (!res.ok) throw new Error('Lỗi lưu file lên máy chủ');

      playSound('correct');
      showToast(`✨ Đã cập nhật thành công "${filename}"!`);
      // Force reload image
      setTimeout(() => {
        window.location.reload();
      }, 1000);
    } catch (err: any) {
      console.error(err);
      alert('Lỗi xử lý ảnh: ' + err.message);
    } finally {
      setIsProcessing(false);
    }
  };

  // Clipboard Paste listener (Ctrl+V)
  useEffect(() => {
    if (!isOpen) return;

    const handlePaste = (e: ClipboardEvent) => {
      const items = e.clipboardData?.items;
      if (!items) return;

      for (let i = 0; i < items.length; i++) {
        if (items[i].type.indexOf('image') !== -1) {
          const blob = items[i].getAsFile();
          if (blob) {
            e.preventDefault();
            processAndUploadImage(blob, selectedAsset.filename);
            return;
          }
        }
      }
    };

    window.addEventListener('paste', handlePaste);
    return () => window.removeEventListener('paste', handlePaste);
  }, [isOpen, selectedAsset]);

  if (!isOpen) return null;

  return (
    <div className="fixed inset-0 z-50 bg-slate-900/70 backdrop-blur-sm flex items-center justify-center p-3 sm:p-5 animate-in fade-in duration-200">
      <div className="bg-white w-full max-w-4xl rounded-3xl shadow-2xl border-2 border-slate-200 flex flex-col max-h-[90vh] overflow-hidden">
        
        {/* Header */}
        <div className="px-6 py-4 border-b border-slate-200 flex items-center justify-between bg-gradient-to-r from-sky-50 to-indigo-50">
          <div className="flex items-center space-x-3">
            <div className="w-10 h-10 rounded-2xl bg-[#009FE3] text-white flex items-center justify-center shadow-md">
              <Sparkles className="w-5 h-5" />
            </div>
            <div>
              <h3 className="text-base font-black text-slate-900">
                Vikoda 3D Asset Studio • Tích Hợp Gemini Pro
              </h3>
              <p className="text-xs text-slate-500 font-medium">
                Copy prompt sang Gemini Pro → Nhấn Ctrl+V để dán ảnh vào đây (Tự động tách nền trong suốt 100%)
              </p>
            </div>
          </div>
          <button
            onClick={onClose}
            className="p-2 rounded-full text-slate-400 hover:text-slate-700 hover:bg-white transition-all cursor-pointer shadow-xs"
          >
            <X className="w-5 h-5" />
          </button>
        </div>

        {/* Toast Alert */}
        {toastMessage && (
          <div className="bg-emerald-600 text-white text-xs font-black px-4 py-2 text-center animate-in fade-in slide-in-from-top-2">
            {toastMessage}
          </div>
        )}

        {/* Content Body: 2 Columns */}
        <div className="flex-1 overflow-hidden flex flex-col md:flex-row">
          
          {/* Left Column: Asset Selection List */}
          <div className="w-full md:w-80 border-r border-slate-200 overflow-y-auto p-3 space-y-2 bg-slate-50/50">
            <div className="text-[11px] font-black uppercase text-slate-400 tracking-wider px-2 py-1">
              Danh sách Asset Game (8 Mục)
            </div>
            {ASSET_DEFINITIONS.map(asset => {
              const isSelected = selectedAssetId === asset.id;
              return (
                <div
                  key={asset.id}
                  onClick={() => {
                    playSound('click');
                    setSelectedAssetId(asset.id);
                  }}
                  className={`p-3 rounded-2xl transition-all cursor-pointer flex items-center gap-3 border ${
                    isSelected
                      ? 'bg-white border-[#009FE3] shadow-md border-b-4'
                      : 'bg-white/80 border-slate-200 hover:bg-white hover:border-slate-300'
                  }`}
                >
                  <div className="w-12 h-12 rounded-xl bg-slate-100 flex items-center justify-center p-1 shrink-0 border border-slate-200 overflow-hidden">
                    <img 
                      src={`${asset.currentSrc}?t=${Date.now()}`} 
                      alt={asset.nameVi} 
                      className="w-full h-full object-contain"
                    />
                  </div>
                  <div className="min-w-0 flex-1">
                    <h4 className="text-xs font-black text-slate-800 truncate">
                      {asset.nameVi}
                    </h4>
                    <p className="text-[10px] text-slate-400 font-mono truncate">
                      {asset.filename}
                    </p>
                  </div>
                </div>
              );
            })}
          </div>

          {/* Right Column: Active Asset Detail & Instant Uploader */}
          <div className="flex-1 overflow-y-auto p-6 space-y-5 flex flex-col justify-between">
            <div className="space-y-4">
              
              <div className="flex items-start justify-between gap-4">
                <div>
                  <span className="text-[10px] font-black uppercase tracking-wider text-[#0070D1] bg-sky-100 px-2.5 py-0.5 rounded-full">
                    Đang chọn
                  </span>
                  <h2 className="text-lg font-black text-slate-900 mt-1">
                    {selectedAsset.nameVi}
                  </h2>
                  <p className="text-xs text-slate-500 font-medium">
                    {selectedAsset.description}
                  </p>
                </div>

                {/* Current Asset Thumbnail */}
                <div className="w-20 h-20 rounded-2xl bg-slate-100 border-2 border-slate-200 p-2 flex items-center justify-center shrink-0 shadow-inner">
                  <img 
                    src={`${selectedAsset.currentSrc}?t=${Date.now()}`} 
                    alt="Current Asset"
                    className="w-full h-full object-contain filter drop-shadow-sm" 
                  />
                </div>
              </div>

              {/* Step 1: Prompt Box */}
              <div className="space-y-1.5">
                <div className="flex items-center justify-between">
                  <label className="text-xs font-black text-slate-700 flex items-center gap-1.5">
                    <span className="w-5 h-5 rounded-full bg-[#009FE3] text-white flex items-center justify-center text-[10px]">1</span>
                    Prompt Chuẩn Cho Gemini Pro (Tiếng Anh)
                  </label>
                  <button
                    onClick={() => copyPrompt(selectedAsset)}
                    className="px-3 py-1.5 rounded-xl bg-[#009FE3] hover:bg-[#0077b6] text-white text-xs font-black flex items-center gap-1.5 transition-all cursor-pointer shadow-xs active:scale-95"
                  >
                    {copiedId === selectedAsset.id ? <Check className="w-3.5 h-3.5" /> : <Copy className="w-3.5 h-3.5" />}
                    <span>{copiedId === selectedAsset.id ? 'Đã Copy!' : 'Copy Prompt'}</span>
                  </button>
                </div>
                <div className="p-3 bg-slate-50 border border-slate-200 rounded-2xl text-xs font-mono text-slate-700 select-all leading-relaxed max-h-32 overflow-y-auto">
                  {selectedAsset.prompt}
                </div>
              </div>

              {/* Step 2: Instant Paste / Upload Dropzone */}
              <div className="space-y-1.5">
                <label className="text-xs font-black text-slate-700 flex items-center gap-1.5">
                  <span className="w-5 h-5 rounded-full bg-emerald-600 text-white flex items-center justify-center text-[10px]">2</span>
                  Nhận Ảnh Mới (Copy từ Gemini & Nhấn Ctrl+V ở đây)
                </label>

                <div 
                  onClick={() => fileInputRef.current?.click()}
                  className="border-2 border-dashed border-sky-300 hover:border-[#009FE3] bg-sky-50/50 hover:bg-sky-50 rounded-3xl p-6 text-center cursor-pointer transition-all space-y-2 group"
                >
                  <input 
                    type="file" 
                    ref={fileInputRef} 
                    className="hidden" 
                    accept="image/*"
                    onChange={(e) => {
                      const file = e.target.files?.[0];
                      if (file) processAndUploadImage(file, selectedAsset.filename);
                    }}
                  />

                  <div className="w-12 h-12 rounded-2xl bg-white shadow-md mx-auto flex items-center justify-center text-[#009FE3] group-hover:scale-110 transition-transform">
                    {isProcessing ? (
                      <RefreshCw className="w-6 h-6 animate-spin" />
                    ) : (
                      <Upload className="w-6 h-6" />
                    )}
                  </div>

                  <div>
                    <div className="text-xs font-black text-slate-800">
                      {isProcessing ? 'Đang tự động tách nền & lưu asset...' : 'Bấm để chọn file ảnh hoặc nhấn Ctrl+V trực tiếp'}
                    </div>
                    <div className="text-[11px] text-slate-500 font-medium mt-0.5">
                      Hỗ trợ PNG, JPG, WebP. Thuật toán tự động xóa nền trắng sang trong suốt chuẩn 100%.
                    </div>
                  </div>
                </div>
              </div>

            </div>

            {/* Quick Helper Banner */}
            <div className="p-3 bg-amber-50 border border-amber-200 rounded-2xl text-[11px] text-amber-900 font-medium flex items-center gap-2">
              <span className="text-base">💡</span>
              <span>
                <strong>Cách nhanh nhất:</strong> Trên Gemini Pro, bạn chỉ cần bấm chuột phải vào ảnh → chọn <em>"Sao chép hình ảnh"</em> → rồi quay lại đây bấm <strong>Ctrl + V</strong>!
              </span>
            </div>

          </div>

        </div>

      </div>
    </div>
  );
};
